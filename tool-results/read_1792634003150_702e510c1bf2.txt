   128→   128→export default function HalamanCuaca() {
   129→   129→  const [negeriId, setNegeriId] = useState("selangor");
   130→   130→  const negeri = negeriById(negeriId);
   131→   131→
   132→   132→  const [dataIPU, setDataIPU] = useState<ResponIPU | null>(null);
   133→   133→  const [dataCuaca, setDataCuaca] = useState<ResponCuaca | null>(null);
   134→   134→  const [muatIPU, setMuatIPU] = useState(true);
   135→   135→  const [muatCuaca, setMuatCuaca] = useState(true);
   136→   136→  const [ralatIPU, setRalatIPU] = useState<string | null>(null);
   137→   137→  const [ralatCuaca, setRalatCuaca] = useState<string | null>(null);
   138→   138→  const [segarkan, setSegarkan] = useState(false);
   139→   139→  const [masaKemasKini, setMasaKemasKini] = useState<string | null>(null);
   140→   140→
   141→   141→  // ---------- pemuatan data ----------
   142→   142→  const muatIPUSekarang = useCallback(async (senyap = false) => {
   143→   143→    if (!senyap) setMuatIPU(true);
   144→   144→    setRalatIPU(null);
   145→   145→    try {
   146→   146→      const res = await fetch("/api/ipu", { cache: "no-store" });
   147→   147→      const json = (await res.json()) as ResponIPU;
   148→   148→      if (!res.ok || !json.ok) throw new Error(json.ralat ?? "HTTP ralat");
   149→   149→      setDataIPU(json);
   150→   150→      setMasaKemasKini(json.updatedAt);
   151→   151→    } catch (e) {
   152→   152→      setRalatIPU(e instanceof Error ? e.message : "Gagal memuatkan data IPU");
   153→   153→    } finally {
   154→   154→      setMuatIPU(false);
   155→   155→    }
   156→   156→  }, []);
   157→   157→
   158→   158→  const muatCuacaSekarang = useCallback(
   159→   159→    async (senyap = false) => {
   160→   160→      if (!senyap) setMuatCuaca(true);
   161→   161→      setRalatCuaca(null);

   163→   163→        const res = await fetch(`/api/weather?lat=${negeri.lat}&lon=${negeri.lon}`, {
   164→   164→          cache: "no-store",
   165→   165→        });
   166→   166→        const json = (await res.json()) as ResponCuaca;
   167→   167→        if (!res.ok || !json.ok) throw new Error(json.ralat ?? "HTTP ralat");
   168→   168→        setDataCuaca(json);
   169→   169→      } catch (e) {
   170→   170→        setRalatCuaca(e instanceof Error ? e.message : "Gagal memuatkan data cuaca");
   171→   171→      } finally {
   172→   172→        setMuatCuaca(false);
   173→   173→      }
   174→   174→    },
   175→   175→    [negeri.lat, negeri.lon],
   176→   176→  );
   177→   177→
   178→   178→  // Muat apabila negeri bertukar

   180→   180→    muatIPUSekarang();

   182→   182→  }, [muatIPUSekarang, muatCuacaSekarang]);

   212→   212→        id: s.id,
   213→   213→        tempat: info.tempat,
   214→   214→        daerah: info.daerah,
   215→   215→        ipu: s.ipu,
   216→   216→        pencemar: s.pencemar,
   217→   217→        masaBacaan: s.masaBacaan,
   218→   218→      });
   219→   219→      peta.set(info.daerah, senaraiDlm);
   220→   220→    }
   221→   221→    const kumpulan: KumpulanDaerah[] = [];

   228→   228→        daerah,
   229→   229→        stesen: [...stesen].sort((a, b) => (b.ipu ?? -1) - (a.ipu ?? -1)),
   230→   230→        maks,
   231→   231→        tahap: tahapDariNilai(maks >= 0 ? maks : null),
   232→   232→      });
   233→   233→    }

   246→   246→  const labelTahap = info?.label ?? "Tiada Bacaan";
   247→   247→  const bilDaerah = kumpulanDaerah.length;
   248→   248→  const bilTerjejas = kumpulanDaerah.filter((k) => k.maks > 50).length;
   249→   249→  const bilStesen = kumpulanDaerah.reduce((n, k) => n + k.stesen.length, 0);
   250→   250→
   251→   251→  const cuacaSekarang = dataCuaca?.current ?? null;
   252→   252→  const infoCuacaSekarang = infoKodCuaca(cuacaSekarang?.kod);
   253→   253→  const hariIni = dataCuaca?.daily?.[0] ?? null;
   254→   254→
   255→   255→  const anim = (delay = 0) => ({
   256→   256→    initial: { opacity: 0, y: 16 },
   257→   257→    animate: { opacity: 1, y: 0 },
   258→   258→    transition: { duration: 0.45, delay, ease: "easeOut" as const },
   259→   259→  });
   260→   260→
   261→   261→  return (
   262→   262→    <div
   263→   263→      className="relative min-h-screen w-full overflow-x-clip"
   264→   264→      style={{
   265→   265→        background:
   266→   266→          "linear-gradient(168deg, #FFF4E3 0%, #FFE8CC 34%, #FFD8B2 62%, #FFC48E 88%, #FBB878 100%)",
   267→   267→      }}
   268→   268→    >
   269→   269→      {/* Hiasan latar lembut */}
   270→   270→      <div
   271→   271→        aria-hidden="true"
   272→   272→        className="pointer-events-none absolute inset-0 overflow-hidden"
   273→   273→      >
   274→   274→        <div
   275→   275→          className="absolute -top-24 right-[-10%] h-72 w-72 rounded-full opacity-60 blur-3xl sm:h-96 sm:w-96"
   276→   276→          style={{ background: "radial-gradient(circle, #FFD9A8 0%, rgba(255,217,168,0) 70%)" }}
   277→   277→        />
   278→   278→        <div
   279→   279→          className="absolute bottom-[-8%] left-[-12%] h-80 w-80 rounded-full opacity-50 blur-3xl"
   280→   280→          style={{ background: "radial-gradient(circle, #FFE9C9 0%, rgba(255,233,201,0) 70%)" }}
   281→   281→        />
   282→   282→      </div>
   283→   283→
   284→   284→      <div className="relative z-10 mx-auto flex min-h-screen w-full max-w-md flex-col px-4 pb-6 pt-6 sm:max-w-xl sm:px-6 lg:max-w-5xl lg:px-8">
   285→   285→        {/* ================= HEADER ================= */}
   286→   286→        <motion.header {...anim(0)} className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   287→   287→          <div>
   288→   288→            <p className="text-[11px] font-bold uppercase tracking-[0.22em] text-[#B0723A]">
   289→   289→              Perak &amp; Selangor · Masa Nyata
   290→   290→            </p>
   291→   291→            <h1 className="mt-0.5 text-[26px] font-bold leading-tight text-[#45220D] sm:text-3xl">
   292→   292→              Cuaca &amp; Jerebu
   293→   293→            </h1>
   294→   294→          </div>
   295→   295→          <div className="flex items-center gap-2">
   296→   296→            <div className="relative flex-1 sm:flex-none sm:w-[228px]">
   297→   297→              <MapPin

   315→   315→                      className="rounded-xl text-[15px] text-[#45220D] focus:bg-[#FFE8CC] data-[highlighted]:bg-[#FFE8CC]"

   317→   317→                      {l.nama}
